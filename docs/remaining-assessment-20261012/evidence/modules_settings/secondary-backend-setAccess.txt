  178   /**
  179    * Apply the same ACL to every file currently carrying `folder` (or the
  180    * unfiled group when `folder` is null). Folders are labels on attachments —
  181    * there is no folder table — so "folder access" is bulk file access.

  183   async setFolderAccess(
  184     ctx: TenantContext,
  185     clientId: string,
  186     folder: string | null,
  187     viewerMode: ViewerMode,
  188     viewerUserIds: string[],
  189   ) {
  190     const normalizedFolder = folder?.trim() || null;
  191     const normalizedIds =
  192       viewerMode === 'all' ? [] : [...new Set(viewerUserIds.map((id) => id.trim()).filter(Boolean))];
  193 
  194     return this.prisma.withTenant(ctx.tenantId, async (tx) => {
  195       if (normalizedIds.length) {
  196         await this.assertTenantMembers(tx, ctx.tenantId, normalizedIds);
  197       }
  198 
  199       const result = await tx.engagementAttachment.updateMany({
  200         where: { prospectId: null,
  201           tenantId: ctx.tenantId,
  202           clientId,
  203           folder: normalizedFolder,
  204           // Only rewrite files the caller can already see — never a vehicle for
  205           // changing ACLs on restricted files the caller is not on.
  206           OR: this.visibleWhere(ctx.userId),
  207         },
  208         data: { viewerMode, viewerUserIds: normalizedIds },
  209       });
  210       return {
  211         folder: normalizedFolder,
  212         viewerMode,
  213         viewerUserIds: normalizedIds,
  214         updated: result.count,
  215       };
  216     });
  217   }

  219   /** Prisma `OR` clause: all-mode files, or restricted files listing the user / uploaded by them. */
  220   private visibleWhere(userId: string) {
  221     return [
  222       { viewerMode: 'all' },
  223       { viewerMode: 'restricted', viewerUserIds: { has: userId } },
  224       { uploadedByUserId: userId },
  225     ];
  226   }

  228   private canView(
  229     userId: string,
  230     row: { viewerMode: string; viewerUserIds: string[]; uploadedByUserId: string | null },
  231   ): boolean {
  232     if (row.viewerMode !== 'restricted') return true;
  233     if (row.uploadedByUserId && row.uploadedByUserId === userId) return true;
  234     return row.viewerUserIds.includes(userId);
  235   }
