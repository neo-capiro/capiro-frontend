  139   async setAccess(

  143     viewerMode: ViewerMode,
  144     viewerUserIds: string[],
  145   ) {

  148

  150       const existing = await tx.engagementAttachment.findFirst({
  151         where: { prospectId: null, id: attachmentId, tenantId: ctx.tenantId, clientId },
  152         select: { id: true, viewerMode: true, viewerUserIds: true, uploadedByUserId: true },
  153       });
  154       if (!existing) throw new NotFoundException('File not found for this client');
  155       if (!this.canView(ctx.userId, existing)) {
  156         throw new NotFoundException('File not found for this client');
  157       }
  158 
  159       if (normalizedIds.length) {
  160         await this.assertTenantMembers(tx, ctx.tenantId, normalizedIds);
  161       }
  162 
  163       return tx.engagementAttachment.update({
  164         where: { id: attachmentId },
  165         data: { viewerMode, viewerUserIds: normalizedIds },

  167           id: true,
  168           fileName: true,
