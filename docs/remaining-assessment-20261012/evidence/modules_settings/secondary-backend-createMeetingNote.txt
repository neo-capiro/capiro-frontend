 5798       }
 5799       assertExpectedRevision(input.expectedRevision, note.updatedAt, 'meeting_note', noteId, {
 5800         id: note.id,
 5801         status: note.status,
 5802         confidential: note.confidential,
 5803         accessLevel: note.accessLevel,
 5804       });
 5805 
 5806       return tx.meetingNote.update({
 5807         where: { id: noteId },
 5808         data: {
 5809           bodyCiphertext: encrypted.bodyCiphertext,
 5810           iv: encrypted.iv,
 5811           authTag: encrypted.authTag,
 5812           keyVersion: encrypted.keyVersion,
 5813           // Phase 2 §d: a manual edit moves an approved/draft note back to
 5814           // `edited` (mirrors updateMeetingDebrief/updateMeetingPrep) and

 5818           status: 'edited',
 5819           approvedAt: null,
 5820           approvedByUserId: null,
 5821           sharedAt: null,
 5822           sharedByUserId: null,

 5824           ...(input.accessLevel === undefined ? {} : { accessLevel: input.accessLevel }),
 5825         },
 5826         select: noteMetadataSelect(),
